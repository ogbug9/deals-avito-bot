# Мусор: объявление остаётся в базе, но помечается, не участвует в оценке рынка и не идёт в кандидаты.
# Формат:  причина = где : шаблон|шаблон    (где: title — только в названии, any — в названии и описании)
# Причины: box — коробка без товара, incomplete — нет деталей (некомплект), scrap — утиль, на запчасти.
# Шаблон — регулярное выражение, регистр не важен.

box = title : ^[^a-zа-я0-9]*(пуст\w+\s+)?коробк\w*\s+(от|для|из[\s-]под)|упаковк\w*\s+от|только\s+коробк|коробк\w*\s+без\s+(карт|видеокарт)
box = any : пустая\s+коробка|коробка\s+без\s+(карты|видеокарты|содержимого)|продам\s+коробку\s+от

incomplete = any : без\s+(систем\w+\s+)?охлаждения|охлаждение\s+отсутствует|без\s+кулер(а|ов)\b|без\s+радиатора
incomplete = any : снят\w*\s+(штатн\w+\s+)?(систем\w+\s+охлаждения|охлаждение|радиатор|кулер)|только\s+плата|плата\s+без
incomplete = any : без\s+(видео)?чипа|без\s+(всех\s+)?комплектующих|без\s+начинки|без\s+железа|пуст\w+\s+корпус|корпус\s+без\s+(комплектующих|начинки)
incomplete = title : ^[^a-zа-я0-9]*(охлаждение|радиатор|кулер\w*|вентилятор\w*|кожух|бэкплейт|backplate|плата)\s+(от|для|с)\s

scrap = any : утил(ь|я|ю|е|изац)|в\s+утил|на\s+разбор|на\s+детали|донор|под\s+восстановлени|на\s+запчаст|под\s+запчаст|в\s+ремонт\s+или
